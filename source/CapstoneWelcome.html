Problem statement

Title: Capstone welcome after successful forward

If there is no session, send the user back to login. If there is a session, greet the user.

What you must do:

  1. Use getSession(false). Guard against a missing session.
  2. Print Welcome , explain forward / session / cookie, and give a Logout link.
  3. Write complete Java and complete web.xml.

Capstone project source (Java + web.xml)

1) Java - CapstoneWelcome.java

import java.io.IOException;
import java.io.PrintWriter;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import jakarta.servlet.http.HttpSession;

/** Capstone welcome page after successful forward. */
public class CapstoneWelcome extends HttpServlet {

    private static final long serialVersionUID = 1L;

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        show(request, response);
    }

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        show(request, response);
    }

    private void show(HttpServletRequest request, HttpServletResponse response) throws IOException {
        response.setContentType("text/html");
        PrintWriter out = response.getWriter();

        HttpSession session = request.getSession(false);
        if (session == null || session.getAttribute("username") == null) {
            response.sendRedirect("capstone.html");
            return;
        }

        String user = (String) session.getAttribute("username");
        out.println("<!DOCTYPE html>");
        out.println("<html><head><meta charset='UTF-8'><title>Capstone Welcome</title>");
        out.println("<link rel='stylesheet' href='portal.css'></head><body><div class='wrap'>");
        out.println("<h1>Welcome " + user + "</h1>");
        out.println("<div class='card'>");
        out.println("<p>Login success using <b>forward</b>.</p>");
        out.println("<p>Username stored in <b>HttpSession</b>.</p>");
        out.println("<p>Cookie <b>remember</b> also created for next visit.</p>");
        out.println("<p><a class='btn' href='CapstoneLogout'>Logout</a></p>");
        out.println("</div>");
        out.println("<p><a href='index.html#ch-capstone'>Return to Index</a></p>");
        out.println("</div></body></html>");
    }
}
