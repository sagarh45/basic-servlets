Problem statement (ICA 3 · mail / payment result page)

Title: TargetServlet shows FORWARD (attribute) or REDIRECT (query)

After Gmail forward, getAttribute("message") is present and the URL is still /forward. After Amazon Pay redirect, only query from= is present and the URL is /target.

What you must do:

  1. Handle GET (redirect) and POST (forward keeps POST).
  2. If attribute message exists, print FORWARD. Else if from exists, print REDIRECT.

1) Java code - TargetServlet.java

import java.io.IOException;
import java.io.PrintWriter;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

/**
 * Target servlet for Forward and Redirect demos.
 */
public class TargetServlet extends HttpServlet {

    private static final long serialVersionUID = 1L;

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        response.setContentType("text/html");
        PrintWriter out = response.getWriter();

        // Check if request came via forward (attribute) or redirect (query param)
        String forwardedMessage = (String) request.getAttribute("message");
        String forwardedSource = (String) request.getAttribute("source");
        String redirectFrom = request.getParameter("from");

        out.println("<!DOCTYPE html>");
        out.println("<html><head><title>Target Servlet</title></head><body>");
        out.println("<h2>TargetServlet - Destination Page</h2>");

        if (forwardedMessage != null) {
            out.println("<p style='color:green;'><b>Arrived via FORWARD from " + forwardedSource + "</b></p>");
            out.println("<p>Message: " + forwardedMessage + "</p>");
            out.println("<p><i>Notice: Browser URL still shows /forward (forward keeps same request).</i></p>");
        } else if (redirectFrom != null) {
            out.println("<p style='color:blue;'><b>Arrived via REDIRECT from " + redirectFrom + "</b></p>");
            out.println("<p><i>Notice: Browser URL changed to /target (redirect creates new request).</i></p>");
        } else {
            out.println("<p>Direct access to TargetServlet.</p>");
        }

        out.println("<p><a href='forward'>Try Forward</a> | <a href='redirect'>Try Redirect</a> | <a href='index.html#ex-target'>Index</a></p>");
        out.println("</body></html>");

        System.out.println("[TargetServlet] doGet() executed.");
    }

    // Forward from a POST form keeps POST, so TargetServlet must also handle doPost.
    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        doGet(request, response);
    }
}
